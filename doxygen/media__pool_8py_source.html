490 elif tool_name == "get_folder_clips":
491 return client.get_folder_clips(arguments.get("folder_path", ""))
492 elif tool_name == "get_folder_subfolders":
493 return client.get_folder_subfolders(arguments.get("folder_path", ""))
494 elif tool_name == "get_selected_pool_clips":
495 return client.get_selected_pool_clips()
496 elif tool_name == "set_selected_pool_clip":
497 result = client.set_selected_pool_clip(arguments.get("clip_id", ""))
498 return "Clip selected" if result else "Selection failed"

500 result = client.refresh_media_pool_folders()
501 return "Folders refreshed" if result else "Refresh failed"
502

504 elif tool_name == "create_media_pool_folder":
505 fp = arguments.get("folder_path", "")
506 name = arguments.get("name", "")
507 result = client.create_media_pool_folder(fp, name)
508 return f"Created folder '{name}' in '{fp}'" if result else "Failed"
509 elif tool_name == "delete_media_pool_folders":
510 fps = arguments.get("folder_paths", [])

514 ):
515 return err
516 result = client.delete_media_pool_folders(fps)
517 return f"Deleted {len(fps)} folder(s)" if result else "Delete failed"
518 elif tool_name == "move_clips_to_folder":
519 clip_ids = arguments.get("clip_ids", [])
520 target = arguments.get("target_folder_path", "")
521 if err := _confirm_gate(
522 arguments, f"Moves {len(clip_ids)} clip(s) to '{target}'."

527 f"Moved {len(clip_ids)} clip(s) to '{target}'"

532 fps = arguments.get("folder_paths", [])
533 target = arguments.get("target_folder_path", "")
