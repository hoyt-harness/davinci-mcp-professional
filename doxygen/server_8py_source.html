39class DaVinciMCPServer:
40 """
41 DaVinci Resolve MCP Server.
42
43 Provides a clean interface between MCP clients and DaVinci Resolve
44 through a kernel + domain architecture. The kernel exposes a fixed
45 set of tools at session start; domain tools are registered dynamically
46 via activate_domain.
47 """

62 def _rebuild_routing_tables(self) -> None:
63 """Rebuild tool→domain maps from the current active-domains set.
64
65 Called once on __init__ and after every activation/deactivation.
66 Not called on every tool call — the maps are stable between changes.
67 """
68 self._tool_to_domain = {
69 tool.name: domain_name
70 for domain_name, domain in self._active_domains.items()
71 for tool in domain.get_tools()
72 }
73 self._inactive_tool_to_domain = {
74 tool.name: domain_name
75 for domain_name, domain in DOMAIN_REGISTRY.items()
76 if domain_name not in self._active_domains
77 for tool in domain.get_tools()
78 }

84 async def _activate_domain(self, domain_name: str) -> dict[str, Any]:
85 if domain_name not in DOMAIN_REGISTRY:
86 return {
87 "error": (
88 f"Unknown domain: '{domain_name}'. "
89 f"Available: {sorted(DOMAIN_REGISTRY)}"
90 )
91 }
92 if domain_name in self._active_domains:
93 return {"status": "already_active", "domain": domain_name}
94
95 self._active_domains[domain_name] = DOMAIN_REGISTRY[domain_name]
96 self._rebuild_routing_tables()
97 await request_ctx.get().session.send_tool_list_changed()
98 return {
99 "status": "activated",
100 "domain": domain_name,
101 "tools_added": [t.name for t in DOMAIN_REGISTRY[domain_name].get_tools()],
102 }
103
104 async def _deactivate_domain(self, domain_name: str) -> dict[str, Any]:
105 if domain_name not in self._active_domains:
106 return {
107 "error": (
108 f"Domain '{domain_name}' is not active. "
109 f"Active domains: {sorted(self._active_domains)}"
110 )
111 }
112 del self._active_domains[domain_name]
113 self._rebuild_routing_tables()
114 await request_ctx.get().session.send_tool_list_changed()
115 return {"status": "deactivated", "domain": domain_name}
116
117 def _list_domains(self) -> list[dict[str, Any]]:
118 return [
119 {
120 "name": name,
121 "description": domain.description,
122 "active": name in self._active_domains,
123 "tool_count": len(domain.get_tools()),
124 }
125 for name, domain in DOMAIN_REGISTRY.items()
126 ]

132 async def _call_kernel_tool(self, name: str, arguments: dict[str, Any]) -> Any:
133 if name == "activate_domain":
134 return await self._activate_domain(arguments.get("domain", ""))
135 elif name == "deactivate_domain":
136 return await self._deactivate_domain(arguments.get("domain", ""))
137 elif name == "list_domains":
138 return self._list_domains()
139 elif name == "get_version":
140 return self.resolve_client.get_version()
141 elif name == "get_current_page":
142 return self.resolve_client.get_current_page()
143 elif name == "switch_page":
144 return self.resolve_client.switch_page(arguments.get("page", ""))
145 return f"Unknown kernel tool: {name}"
146
147 async def _dispatch_tool(self, name: str, arguments: dict[str, Any]) -> Any:
148 """Route a tool call through the four-case dispatch (FR-008)."""
149 if name in _KERNEL_TOOLS:
150 return await self._call_kernel_tool(name, arguments)
151 elif name in self._tool_to_domain:
152 domain = self._active_domains[self._tool_to_domain[name]]
153 return await domain.dispatch(name, arguments, self.resolve_client)
154 elif name in self._inactive_tool_to_domain:
155 domain_name = self._inactive_tool_to_domain[name]
156 return (
157 f"Tool '{name}' is in domain '{domain_name}'. "
158 f"Call activate_domain('{domain_name}') first."
159 )
160 else:
161 return f"Unknown tool: {name}"

178 def _register_handlers(self) -> None:
179 """Register MCP server handlers."""
180
181 @self.server.list_tools()
182 async def handle_list_tools() -> list[types.Tool]: # type: ignore
183 return await self._handle_list_tools()
184
185 @self.server.call_tool()
186 async def handle_call_tool( # type: ignore
187 name: str, arguments: dict[str, Any] | None = None
188 ) -> list[types.TextContent]:
189 if arguments is None:
190 arguments = {}
191 try:
192 if name not in _KERNEL_TOOLS or name in (
193 "get_version",
194 "get_current_page",
195 "switch_page",
196 ):
197 if not self.resolve_client.is_connected():
198 self.resolve_client.connect()
199 result = await self._dispatch_tool(name, arguments)
200 return [types.TextContent(type="text", text=str(result))]
201 except DaVinciResolveError as e:
202 error_msg = f"DaVinci Resolve error: {e}"
203 logger.exception(error_msg)
204 return [types.TextContent(type="text", text=error_msg)]
205 except Exception as e:
206 error_msg = f"Unexpected error: {e}"
207 logger.exception(error_msg)
208 return [types.TextContent(type="text", text=error_msg)]

236 async def _read_resource(self, uri: str) -> Any:
237 """Dispatch a resource read to the resolve client."""
238 if uri == "resolve://version":
239 return self.resolve_client.get_version()
240 elif uri == "resolve://current-page":
241 return self.resolve_client.get_current_page()
242 elif uri == "resolve://projects":
243 return self.resolve_client.list_projects()
244 elif uri == "resolve://current-project":
245 name = self.resolve_client.get_current_project_name()
246 return name if name else "No project open"
247 elif uri == "resolve://timelines":
248 return self.resolve_client.list_timelines()
249 elif uri == "resolve://current-timeline":
250 name = self.resolve_client.get_current_timeline_name()
251 return name if name else "No timeline active"
252 elif uri == "resolve://media-clips":
253 return self.resolve_client.list_media_clips()
254 else:
255 raise ValueError(f"Unknown resource: {uri}")
