57 def connect(self) -> None:
58 """Connect to DaVinci Resolve."""
59 # Check if Resolve is running
60 if not check_resolve_running():
61 raise DaVinciResolveNotRunningError(
62 "DaVinci Resolve is not running. Please start DaVinci Resolve first."
63 )
64
65 # Set up environment (env vars, sys.path, DLL search dirs)
66 if not setup_resolve_environment():
67 raise DaVinciResolveConnectionError(
68 "Failed to set up DaVinci Resolve environment variables."
69 )
70
71 # On Windows, fusionscript.dll discovers Python via the Windows
72 # registry and loads python3.dll by full path. If the running
73 # Python is a different installation (e.g. uv-managed), two
74 # runtimes end up in the same process and it crashes.
75 compat_ok, compat_msg = check_python_runtime_compatibility()
76 if not compat_ok:
77 raise DaVinciResolveConnectionError(compat_msg)
78
79 try:
80 # Import and connect to Resolve
81 import DaVinciResolveScript as dvr_script # type: ignore
82
83 self._resolve = dvr_script.scriptapp("Resolve") # type: ignore
84
85 if self._resolve is None:
86 raise DaVinciResolveConnectionError(
87 "Failed to get Resolve object. "
88 "Check that DaVinci Resolve is running."
89 ) # type: ignore
90
91 # Get project manager
92 self._project_manager = self._resolve.GetProjectManager() # type: ignore
93 if self._project_manager is None:
94 raise DaVinciResolveConnectionError("Failed to get Project Manager.")
95
96 # Get current project if one is open
97 self._current_project = ( # type: ignore
98 self._project_manager.GetCurrentProject()
99 )
100
101 self._is_connected = True
102 logger.info(f"Connected to {self.get_version()}")
103
104 except SystemError as e:
105 raise DaVinciResolveConnectionError(
106 f"fusionscript.dll initialization failed: {e}. "
107 "Check that DaVinci Resolve Studio is running with "
108 "external scripting enabled in Preferences."
109 )
110 except ImportError as e:
111 raise DaVinciResolveConnectionError(
112 f"Failed to import DaVinciResolveScript: {e}. "
113 "Check environment variables and DaVinci Resolve installation."
114 )
115 except DaVinciResolveConnectionError:
116 raise
117 except Exception as e:
118 raise DaVinciResolveConnectionError(f"Unexpected error connecting: {e}")

477 def _resolve_timeline_items(
478 self, timeline: Any, item_refs: list[dict[str, Any]]
479 ) -> list[Any]:
480 """Resolve [{track_type, track_index, item_index}] to TimelineItem objects."""
481 items = []
482 for ref in item_refs:
483 track_type = ref.get("track_type", "video")
484 track_index = int(ref.get("track_index", 1))
485 item_index = int(ref.get("item_index", 1))
486 track_items = timeline.GetItemListInTrack(track_type, track_index) or []
487 if item_index < 1 or item_index > len(track_items):
488 raise ValueError(
489 f"item_index {item_index} out of range for {track_type} track {track_index}" # noqa: E501
490 )
491 items.append(track_items[item_index - 1])
492 return items

888 def _get_folder_by_path(self, folder_path: str) -> Any:
889 """Walk the folder tree by slash-separated path from root."""
890 media_pool = self._get_media_pool()
891 folder = media_pool.GetRootFolder()
892 if not folder:
893 raise DaVinciResolveError("Failed to get root folder")
894 if not folder_path or folder_path in ("", "/", "Master"):
895 return folder
896 parts = [p for p in folder_path.strip("/").split("/") if p]
897 for part in parts:
898 subfolders = folder.GetSubFolderList() or []
899 match = next((f for f in subfolders if f.GetName() == part), None)
900 if match is None:
901 raise ValueError(f"Folder not found: '{part}' in path '{folder_path}'")
902 folder = match
903 return folder
904
905 def _get_clip_by_id(self, clip_id: str) -> Any:
906 """Walk the folder tree to find a MediaPoolItem by UUID."""
907 media_pool = self._get_media_pool()
908 root = media_pool.GetRootFolder()
909
910 def _walk(folder: Any) -> Any:
911 for clip in folder.GetClipList() or []:
912 if clip.GetUniqueId() == clip_id:
913 return clip
914 for sub in folder.GetSubFolderList() or []:
915 found = _walk(sub)
916 if found:
917 return found
918 return None
919
920 clip = _walk(root)
921 if clip is None:
922 raise ValueError(f"Clip not found with id '{clip_id}'")
923 return clip
